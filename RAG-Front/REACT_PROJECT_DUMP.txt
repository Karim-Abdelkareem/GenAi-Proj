  import fs from "fs";
  import path from "path";
  
  const IGNORE_DIRS = [
    "node_modules",
    ".git",
    "dist",
    "build",
    ".next",
    "coverage",
  ];
  
  const IGNORE_FILES = [".env", ".env.local", ".env.production"];
  
  const ALLOWED_EXT = [".js", ".jsx", ".ts", ".tsx", ".json", ".css", ".scss"];
  const MAX_FILE_SIZE = 60 * 1024; // 60KB
  
  function walk(dir, depth = 0) {
    let output = "";
    const files = fs.readdirSync(dir);
  
    for (const file of files) {
      if (IGNORE_FILES.includes(file)) continue;
  
      const fullPath = path.join(dir, file);
      const stat = fs.statSync(fullPath);
      const indent = "  ".repeat(depth);
  
      if (stat.isDirectory()) {
        if (IGNORE_DIRS.includes(file)) continue;
        output += `${indent}- 📁 ${file}\n`;
        output += walk(fullPath, depth + 1);
      } else {
        const ext = path.extname(file);
        if (!ALLOWED_EXT.includes(ext)) continue;
  
        output += `${indent}- 📄 ${file}\n`;
  
        if (stat.size <= MAX_FILE_SIZE) {
          let content = fs.readFileSync(fullPath, "utf8");
  
          // 🔒 basic secret masking
          content = content.replace(
            /(apiKey|token|secret|password)\s*[:=]\s*["'][^"']+["']/gi,
            '$1: "***"'
          );
  
          output += `${indent}  ---\n`;
          output += content
            .split("\n")
            .map((l) => indent + "  " + l)
            .join("\n");
          output += `\n${indent}  ---\n`;
        }
      }
    }
  
    return output;
  }
  
  const result = walk(process.cwd());
  fs.writeFileSync("REACT_PROJECT_DUMP.txt", result);

      // ✅ AudioWorklet Processor - Simplified for continuous audio streaming
      // فقط: resampling + PCM conversion + chunking
      
      class RecorderProcessor extends AudioWorkletProcessor {
        constructor() {
          super();
          this.inputSampleRate = 48000; // سيتم تحديثه من main thread
          this.targetSampleRate = 24000;
          this.buffer = []; // ✅ Buffer لتجميع frames
          this.TARGET_SAMPLES = 960; // ✅ 40ms @ 24kHz
          this.MAX_BUFFER_SIZE = 9600; // ✅ حد أقصى: 10 chunks (400ms) لمنع memory leak
      
          // ✅ استقبال الرسائل من main thread
          this.port.onmessage = (e) => {
            if (e.data.type === "init") {
              this.inputSampleRate = e.data.inputSampleRate || 48000;
            } else if (e.data.type === "reset") {
              this.buffer = []; // إعادة تعيين buffer
            }
          };
        }
      
        // ✅ Resample من inputRate إلى outputRate
        downsampleBuffer(buffer, inputRate, outputRate) {
          if (outputRate === inputRate) return buffer;
      
          const ratio = inputRate / outputRate;
          const newLength = Math.round(buffer.length / ratio);
          const result = new Float32Array(newLength);
      
          let offset = 0;
          for (let i = 0; i < newLength; i++) {
            const nextOffset = Math.round((i + 1) * ratio);
            let sum = 0;
            let count = 0;
            for (let j = offset; j < nextOffset && j < buffer.length; j++) {
              sum += buffer[j];
              count++;
            }
            result[i] = count > 0 ? sum / count : 0;
            offset = nextOffset;
          }
          return result;
        }
      
        // ✅ تحويل Float32Array إلى PCM16
        float32ToPCM16(float32Array) {
          const pcm16 = new Int16Array(float32Array.length);
          for (let i = 0; i < float32Array.length; i++) {
            const s = Math.max(-1, Math.min(1, float32Array[i]));
            pcm16[i] = s * 32767;
          }
          return pcm16;
        }
      
        // ✅ تحويل ArrayBuffer إلى base64 (بدون btoa - غير متاح في AudioWorklet)
        // ✅ استخدام base64 encoding يدوي
        arrayBufferToBase64(buffer) {
          const bytes = new Uint8Array(buffer);
          const base64Chars =
            "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
          let result = "";
          let i = 0;
      
          while (i < bytes.length) {
            const a = bytes[i++];
            const b = i < bytes.length ? bytes[i++] : 0;
            const c = i < bytes.length ? bytes[i++] : 0;
      
            const bitmap = (a << 16) | (b << 8) | c;
      
            result += base64Chars.charAt((bitmap >> 18) & 63);
            result += base64Chars.charAt((bitmap >> 12) & 63);
            result +=
              i - 2 < bytes.length ? base64Chars.charAt((bitmap >> 6) & 63) : "=";
            result += i - 1 < bytes.length ? base64Chars.charAt(bitmap & 63) : "=";
          }
          return result;
        }
      
        // eslint-disable-next-line no-unused-vars
        process(inputs, outputs, parameters) {
          const input = inputs[0];
          if (!input || input.length === 0) return true;
      
          const inputChannel = input[0];
          if (!inputChannel || inputChannel.length === 0) return true;
      
          // ✅ Resample من inputRate إلى 24kHz
          const downsampled =
            this.inputSampleRate !== this.targetSampleRate
              ? this.downsampleBuffer(
                  inputChannel,
                  this.inputSampleRate,
                  this.targetSampleRate
                )
              : inputChannel;
      
          // ✅ إضافة كل الـ samples إلى buffer (بدون RMS check على frame صغير)
          for (let i = 0; i < downsampled.length; i++) {
            this.buffer.push(downsampled[i]);
          }
      
          // ✅ منع buffer من النمو المفرط (memory leak protection)
          if (this.buffer.length > this.MAX_BUFFER_SIZE) {
            // ✅ إفراغ buffer إذا تجاوز الحد الأقصى
            this.buffer = [];
            return true;
          }
      
          // ✅ إرسال كل chunk مباشرة (بدون RMS filtering)
          while (this.buffer.length >= this.TARGET_SAMPLES) {
            // ✅ استخراج chunk من buffer
            const chunk = this.buffer.slice(0, this.TARGET_SAMPLES);
            this.buffer = this.buffer.slice(this.TARGET_SAMPLES);
      
            // ✅ حساب RMS (Root Mean Square) لمستوى الصوت
            let sumSquares = 0;
            for (let i = 0; i < chunk.length; i++) {
              sumSquares += chunk[i] * chunk[i];
            }
            const rms = Math.sqrt(sumSquares / chunk.length);
            // ✅ تحويل RMS إلى نسبة مئوية (0-100)
            const volumeLevel = Math.min(100, Math.max(0, rms * 100));
      
            // ✅ تحويل إلى PCM16
            const pcm16 = this.float32ToPCM16(chunk);
      
            // ✅ تحويل إلى base64
            const base64Audio = this.arrayBufferToBase64(pcm16.buffer);
      
            // ✅ إرسال للـ main thread (chunk كامل 960 samples = 40ms)
            // ✅ الموديل يقرر هو الكلام ولا لا
            this.port.postMessage({
              type: "audio",
              audio: base64Audio,
              samples: chunk.length, // ✅ دائماً 960
              volumeLevel: volumeLevel, // ✅ مستوى الصوت (0-100)
            });
          }
      
          return true;
        }
      
        static get parameterDescriptors() {
          return [];
        }
      }
      
      registerProcessor("recorder-processor", RecorderProcessor);

      const BASE_URL = "http://localhost:3000";
      
      export const loginUser = async (email, password) => {
        const response = await fetch(`${BASE_URL}/api/v1/auth/login`, {
          method: "POST",
          headers: {
            "Content-Type": "application/json",
          },
          body: JSON.stringify({ email, password }),
        });
      
        if (!response.ok) {
          const error = await response
            .json()
            .catch(() => ({ message: "Login failed" }));
          throw new Error(error.message || "Login failed");
        }
      
        const data = await response.json();
        return data;
      };

      /**

      import React from "react";
      import { Tag, Timeline, Typography, Space, Tooltip } from "antd";
      import { ArrowUpOutlined, ArrowDownOutlined, MinusOutlined } from "@ant-design/icons";
      
      const { Text } = Typography;
      
      /**
       * DifficultyTracker component
       * @param {object} props - Component props
       * @param {string} props.currentDifficulty - Current difficulty level
       * @param {Array} props.progression - Array of difficulty changes with timestamps
       * @param {boolean} props.showProgression - Whether to show progression timeline
       */
      const DifficultyTracker = ({ currentDifficulty, progression = [], showProgression = false }) => {
        // Difficulty configuration
        const difficultyConfig = {
          beginner: {
            label: "Beginner",
            color: "blue",
            icon: <ArrowDownOutlined />,
          },
          intermediate: {
            label: "Intermediate",
            color: "orange",
            icon: <MinusOutlined />,
          },
          advanced: {
            label: "Advanced",
            color: "red",
            icon: <ArrowUpOutlined />,
          },
        };
      
        const current = difficultyConfig[currentDifficulty] || difficultyConfig.intermediate;
      
        // Format progression for timeline
        const timelineItems = progression.map((item, index) => {
          const prevDifficulty = index > 0 ? progression[index - 1].difficulty : null;
          const config = difficultyConfig[item.difficulty] || difficultyConfig.intermediate;
          const prevConfig = prevDifficulty ? difficultyConfig[prevDifficulty] : null;
      
          let changeIcon = null;
          if (prevConfig) {
            const difficultyOrder = ["beginner", "intermediate", "advanced"];
            const prevIndex = difficultyOrder.indexOf(prevDifficulty);
            const currentIndex = difficultyOrder.indexOf(item.difficulty);
      
            if (currentIndex > prevIndex) {
              changeIcon = <ArrowUpOutlined style={{ color: "#52c41a" }} />;
            } else if (currentIndex < prevIndex) {
              changeIcon = <ArrowDownOutlined style={{ color: "#ff4d4f" }} />;
            } else {
              changeIcon = <MinusOutlined style={{ color: "#faad14" }} />;
            }
          }
      
          return {
            color: config.color,
            children: (
              <Space>
                {changeIcon}
                <Tag color={config.color} icon={config.icon}>
                  {config.label}
                </Tag>
                <Text type="secondary" style={{ fontSize: "12px" }}>
                  {item.timestamp
                    ? new Date(item.timestamp).toLocaleTimeString("en-US", {
                        hour: "2-digit",
                        minute: "2-digit",
                      })
                    : ""}
                </Text>
              </Space>
            ),
          };
        });
      
        return (
          <Space direction="vertical" size="small" style={{ width: "100%" }}>
            {/* Current Difficulty */}
            <Space>
              <Text type="secondary" style={{ fontSize: "12px" }}>
                Current Difficulty:
              </Text>
              <Tooltip title={`Interview difficulty: ${current.label}`}>
                <Tag color={current.color} icon={current.icon} style={{ fontSize: "13px", padding: "4px 12px" }}>
                  {current.label}
                </Tag>
              </Tooltip>
            </Space>
      
            {/* Progression Timeline */}
            {showProgression && progression.length > 0 && (
              <div style={{ marginTop: "8px" }}>
                <Text type="secondary" style={{ fontSize: "12px", marginBottom: "8px", display: "block" }}>
                  Difficulty Progression:
                </Text>
                <Timeline size="small" items={timelineItems} />
              </div>
            )}
          </Space>
        );
      };

      /**
       * InterviewTypeModal Component
       * Modal for selecting interview type (Text or Voice) and voice gender
       * @module components/InterviewTypeModal
       */
      
      import React, { useState } from "react";
      import { Modal, Radio, Space, Typography, Divider } from "antd";
      import { MessageOutlined, AudioOutlined } from "@ant-design/icons";
      
      const { Text, Title } = Typography;
      
      /**
       * Interview Type Selection Modal
       * @param {boolean} visible - Modal visibility
       * @param {Function} onStart - Callback with { type: 'text' | 'voice', voiceGender?: 'male' | 'female' | 'neutral' }
       * @param {Function} onCancel - Cancel callback
       * @returns {JSX.Element} - Modal component
       */
      const InterviewTypeModal = ({ visible, onStart, onCancel }) => {
        const [interviewType, setInterviewType] = useState("text");
        const [voiceGender, setVoiceGender] = useState("neutral");
      
        const handleStart = () => {
          if (interviewType === "voice") {
            onStart({ type: "voice", voiceGender });
          } else {
            onStart({ type: "text" });
          }
          // Reset state
          setInterviewType("text");
          setVoiceGender("neutral");
        };
      
        const handleCancel = () => {
          onCancel();
          // Reset state
          setInterviewType("text");
          setVoiceGender("neutral");
        };
      
        return (
          <Modal
            title="Select Interview Type"
            open={visible}
            onOk={handleStart}
            onCancel={handleCancel}
            okText="Start Interview"
            cancelText="Cancel"
            width={500}
          >
            <Space direction="vertical" size="large" style={{ width: "100%" }}>
              <div>
                <Title level={5}>Choose Interview Mode</Title>
                <Radio.Group
                  value={interviewType}
                  onChange={(e) => setInterviewType(e.target.value)}
                  style={{ width: "100%" }}
                >
                  <Space direction="vertical" size="middle" style={{ width: "100%", marginTop: 12 }}>
                    <Radio value="text" style={{ width: "100%" }}>

      export function useMicrophone() {

              autoGainControl: true, // ✅ مفعّل - AGC + dynamic threshold = ممتاز (AGC تخلي صوتك ثابت والـ threshold بيتحرك معاه)

            "🎙️ [MIC INIT] Requesting microphone with constraints:",
            constraints
          );
      
          const s = await navigator.mediaDevices.getUserMedia(constraints);
      
          // 🔥 Log إعدادات الميكروفون الفعلية
          const audioTracks = s.getAudioTracks();
          if (audioTracks.length > 0) {
            const settings = audioTracks[0].getSettings();
            const settingsData = {
              deviceId: settings.deviceId,
              sampleRate: settings.sampleRate,
              channelCount: settings.channelCount,
              echoCancellation: settings.echoCancellation,
              noiseSuppression: settings.noiseSuppression,
              autoGainControl: settings.autoGainControl,
            };
            console.log("🎙️ [MIC SETTINGS]", settingsData);
          }
      
          setStream(s);
          return s;

        const stopMic = () => {
          if (stream) {
            stream.getTracks().forEach((track) => track.stop());
            setStream(null);

          }
        };

      import { useRef, useState } from "react";
      
      export function useRecorder(send, isAiSpeaking) {
        const [volumeLevel, setVolumeLevel] = useState(0); // ✅ مستوى الصوت (0-100)
        const audioContextRef = useRef(null);
        const processorRef = useRef(null);
        const sourceRef = useRef(null);
        const streamRef = useRef(null);
        const inputSampleRateRef = useRef(48000);
        const isRecordingRef = useRef(false);
        const isPausedRef = useRef(false);
        const silenceTimerRef = useRef(null);
        const workletLoadedRef = useRef(false);
        const pendingEndRef = useRef(false);
      
        const startRecording = async (stream) => {
          // ✅ حفظ الـ stream
          streamRef.current = stream;
      
          // ✅ الحصول على sampleRate الفعلي من stream
          const audioTracks = stream.getAudioTracks();
          if (audioTracks.length > 0) {
            const settings = audioTracks[0].getSettings();
            inputSampleRateRef.current = settings.sampleRate || 48000;
            console.log("🎙️ [SAMPLE RATE] Input:", inputSampleRateRef.current, "Hz");
          }
      
          // ✅ إذا كان التسجيل شغال، لا تفعل شيء
          if (isRecordingRef.current) return;
      
          // ✅ إعادة تعيين flags
          isPausedRef.current = false;
      
          // ✅ إنشاء AudioContext بـ sampleRate native (input sample rate)
          if (!audioContextRef.current) {
            audioContextRef.current = new AudioContext({
              sampleRate: inputSampleRateRef.current, // غالبًا 48000
            });
            console.log(
              "🎙️ [AUDIO CONTEXT] Created with sample rate:",
              audioContextRef.current.sampleRate,
              "Hz"
            );
          }
      
          // ✅ تحميل AudioWorklet processor (مرة واحدة فقط)
          if (!workletLoadedRef.current) {
            try {
              await audioContextRef.current.audioWorklet.addModule(
                "/worklets/recorder-processor.js"
              );
              workletLoadedRef.current = true;
              console.log("✅ [WORKLET LOADED] AudioWorklet processor ready");
            } catch (error) {
              console.error("❌ [WORKLET ERROR]", error);
              throw error;
            }
          }
      
          // ✅ إنشاء MediaStreamSource من الـ stream
          if (!sourceRef.current) {
            sourceRef.current =
              audioContextRef.current.createMediaStreamSource(stream);
          }
      
          // ✅ إنشاء AudioWorkletNode
          if (!processorRef.current) {
            processorRef.current = new AudioWorkletNode(
              audioContextRef.current,
              "recorder-processor"
            );
      
            // ✅ إرسال inputSampleRate إلى processor
            processorRef.current.port.postMessage({
              type: "init",
              inputSampleRate: inputSampleRateRef.current,
            });
      
            // ✅ إعادة تعيين buffer عند البدء
            processorRef.current.port.postMessage({
              type: "reset",
            });
      
            // ✅ استقبال البيانات من processor - إرسال continuous audio
            processorRef.current.port.onmessage = (e) => {
              const msg = e.data;
      
              // ✅ إذا كان AI يتكلم أو التسجيل متوقف، تجاهل الصوت
              if (isPausedRef.current || isAiSpeaking()) {
                return;
              }
      
              if (msg.type === "audio") {
                // ✅ تحديث مستوى الصوت للـ UI
                if (msg.volumeLevel !== undefined) {
                  setVolumeLevel(msg.volumeLevel);
                }
      
                // ✅ إرسال الصوت مباشرة (continuous)
                send({
                  type: "audio",
                  audio: msg.audio,
                });
      
                console.log("🎤 [AUDIO SENT]", {
                  timestamp: new Date().toISOString(),
                  samples: msg.samples,
                  duration: `${(msg.samples / 24000) * 1000}ms`,
                  format: "PCM16@24kHz (AudioWorklet)",
                  volumeLevel: msg.volumeLevel?.toFixed(2) || "N/A",
                });
      
                // ✅ إلغاء silence timer عند إرسال صوت وبدء timer جديد (2 ثواني)
                if (silenceTimerRef.current) {
                  clearTimeout(silenceTimerRef.current);
                  silenceTimerRef.current = null;
                }
      
                // ✅ Silence timer للـ UX فقط (2 ثواني) - عشان الموديل يرد بعد الصمت
                silenceTimerRef.current = setTimeout(() => {
                  // ✅ ممنوع إرسال commit أثناء AI Speaking
                  if (isAiSpeaking()) {
                    silenceTimerRef.current = null;
                    return;
                  }
      
                  if (!pendingEndRef.current) {
                    console.log("🔇 [SILENCE DETECTED] Sending 'commit' signal (UX)");
      
                    try {
                      send({ type: "input_audio_buffer.commit" });
                      pendingEndRef.current = true;
                      console.log(
                        "✅ [COMMIT SENT] 'input_audio_buffer.commit' signal sent to backend (after silence)"
                      );
                    } catch (error) {
                      console.error("❌ [COMMIT SEND ERROR]", error);
                      pendingEndRef.current = false;
                    }
                  }
                  silenceTimerRef.current = null;
                }, 2000); // ✅ 2 ثواني للـ UX
              }
            };
          }
      
          // ✅ ربط AudioNodes
          sourceRef.current.connect(processorRef.current);
          processorRef.current.connect(audioContextRef.current.destination);
      
          isRecordingRef.current = true;
          console.log("▶️ [RECORDING STARTED] Continuous audio streaming");
        };
      
        const pauseRecording = () => {
          // ✅ إيقاف التسجيل عندما AI يتكلم
          if (isPausedRef.current) {
            return;
          }
      
          isPausedRef.current = true;
      
          // ✅ إلغاء silence timer
          if (silenceTimerRef.current) {
            clearTimeout(silenceTimerRef.current);
            silenceTimerRef.current = null;
          }
      
          // ✅ Reset pendingEnd when AI starts speaking (to allow new commit after AI finishes)
          if (pendingEndRef.current) {
            console.log(
              "✅ [RESET PENDING] AI started speaking, resetting pendingEnd"
            );
            pendingEndRef.current = false;
          }
      
          // ✅ إعادة تعيين مستوى الصوت عند pause
          setVolumeLevel(0);
      
          console.log("⏸️ [RECORDING PAUSED] AI is speaking");
        };
      
        const resumeRecording = () => {
          // ✅ إعادة تشغيل التسجيل بعد انتهاء AI
          if (!streamRef.current) {
            console.warn("⚠️ [RESUME FAILED] No stream available");
            return;
          }
      
          if (isRecordingRef.current && !isPausedRef.current) {
            console.log("ℹ️ [RESUME SKIPPED] Already recording");
            return;
          }
      
          // ✅ إعادة تعيين flags
          isPausedRef.current = false;
          pendingEndRef.current = false;
      
          // ✅ إعادة تعيين buffer في processor
          if (processorRef.current) {
            processorRef.current.port.postMessage({
              type: "reset",
            });
          }
      
          console.log("▶️ [RECORDING RESUMED] User can speak now");
        };
      
        // ✅ Reset pendingEnd when AI starts speaking (to allow new commit)
        const onEndAcknowledged = () => {
          // ✅ Note: commit doesn't need ACK, but we reset pendingEnd when AI starts
          if (pendingEndRef.current) {
            console.log("✅ [RESET PENDING] Resetting pendingEnd flag");
            pendingEndRef.current = false;
          }
        };
      
        const stopRecording = () => {
          // ✅ قطع الاتصالات
          if (processorRef.current) {
            processorRef.current.disconnect();
            processorRef.current = null;
          }
      
          if (sourceRef.current) {
            sourceRef.current.disconnect();
            sourceRef.current = null;
          }
      
          if (audioContextRef.current) {
            audioContextRef.current.close();
            audioContextRef.current = null;
            workletLoadedRef.current = false;
          }
      
          // ✅ تنظيف timers
          if (silenceTimerRef.current) {
            clearTimeout(silenceTimerRef.current);
            silenceTimerRef.current = null;
          }
      
          isRecordingRef.current = false;
          streamRef.current = null;
      
          // ✅ إعادة تعيين مستوى الصوت عند إيقاف التسجيل
          setVolumeLevel(0);
        };
      
        return {
          startRecording,
          stopRecording,
          pauseRecording,
          resumeRecording,
          onEndAcknowledged,
          volumeLevel, // ✅ مستوى الصوت للـ UI
        };
      }

              "✅ [WEBSOCKET] Session started (connection = session started)"
            );
          };
      
          socketRef.current.onerror = (error) => {
            console.error("❌ [WEBSOCKET ERROR]", error);
            console.error("❌ [WEBSOCKET] Error details:", {
              type: error.type,
              target: error.target,
              readyState: socketRef.current?.readyState,
            });
          };
      
          socketRef.current.onclose = (event) => {
            console.log("🔌 [WEBSOCKET] Connection closed", {
              code: event.code,
              reason: event.reason,
              wasClean: event.wasClean,
              readyState: socketRef.current?.readyState,
            });
      
            // Reset ref if connection was closed
            if (event.code !== 1000) {
              console.warn("⚠️ [WEBSOCKET] Connection closed unexpectedly");
            }
          };
      
          socketRef.current.onmessage = (event) => {
            const msg = JSON.parse(event.data);
      
            if (msg.type === "audio") {
              aiSpeakingRef.current = true; // GPT بيتكلم
              lastAudioTimeRef.current = performance.now();
      
              // ✅ إلغاء timer القديم
              if (aiDoneTimerRef.current) {
                clearTimeout(aiDoneTimerRef.current);
                aiDoneTimerRef.current = null;
              }
      
              // 🔥 Log الصوت المستقبل من AI
              console.log("🤖 [AI AUDIO RECEIVED]", {
                timestamp: new Date().toISOString(),
                audioLength: msg.audio?.length || 0,
                audioPreview: msg.audio?.substring(0, 50) || "",
              });
      
              callbacksRef.current.onAudio?.(msg.audio);
      
              // ✅ بعد آخر audio chunk، انتظر حتى ينتهي الصوت فعلياً
              aiDoneTimerRef.current = setTimeout(() => {
                if (aiSpeakingRef.current) {
                  // ✅ حساب الوقت المتبقي حتى ينتهي الصوت
                  const remainingTime = getRemainingAudioTime();
      
                  if (remainingTime > 0) {
                    // ✅ انتظار حتى ينتهي الصوت فعلياً
                    setTimeout(() => {
                      if (aiSpeakingRef.current) {
                        aiSpeakingRef.current = false;
                        console.log(
                          "✅ [AI DONE] AI finished speaking (after audio playback)"
                        );
                        callbacksRef.current.onAiDone?.();
                      }
                      aiDoneTimerRef.current = null;
                    }, remainingTime + 100); // +100ms buffer
                  } else {
                    // ✅ الصوت انتهى بالفعل
                    aiSpeakingRef.current = false;
                    console.log("✅ [AI DONE] AI finished speaking (after silence)");
                    callbacksRef.current.onAiDone?.();
                    aiDoneTimerRef.current = null;
                  }
                }
              }, 500); // 500ms بعد آخر audio chunk
            }
      
            if (msg.type === "ai_done") {
              // ✅ إلغاء timer إذا كان موجوداً
              if (aiDoneTimerRef.current) {
                clearTimeout(aiDoneTimerRef.current);
                aiDoneTimerRef.current = null;
              }
      
              // ✅ حساب الوقت المتبقي حتى ينتهي الصوت فعلياً
              const remainingTime = getRemainingAudioTime();
              const delay = remainingTime > 0 ? remainingTime + 200 : 300; // +200ms buffer
      
              setTimeout(() => {
                aiSpeakingRef.current = false;
                console.log(
                  "✅ [AI DONE] AI finished speaking (from backend, after audio playback)"
                );
                callbacksRef.current.onAiDone?.();
              }, delay);
      
              // ❌ لا نستدعي onEndAck() هنا
              // ✅ ai_done = "AI finished speaking" ≠ "User finished speaking"
              // ✅ فقط end_ack يعتبر ACK لـ "end" من المستخدم
            }
      
            // ✅ Note: commit doesn't need ACK - it's fire-and-forget
      
            // ✅ إنهاء المقابلة من backend
            if (msg.type === "interview_ended") {
              console.log("🏁 [INTERVIEW ENDED] Interview ended by AI");
              aiSpeakingRef.current = false;
              callbacksRef.current.onAiDone?.();
              callbacksRef.current.onInterviewEnded?.();
            }
          };
        }, []); // Empty deps - callbacks are stored in ref
      
        const send = useCallback((data) => {
          if (!socketRef.current) {
            console.warn("⚠️ [WEBSOCKET] Cannot send: WebSocket not connected", data);
            return;
          }
      
          const readyState = socketRef.current.readyState;
      
          if (readyState !== WebSocket.OPEN) {
            console.warn(
              "⚠️ [WEBSOCKET] Cannot send: WebSocket not ready. State:",
              readyState,
              data
            );
            return;
          }
      
          try {
            // إرسال audio: { type: "audio", audio: base64 }
            // إرسال commit: { type: "input_audio_buffer.commit" }
            const message = JSON.stringify(data);
            socketRef.current.send(message);
            console.log("📤 [WEBSOCKET SENT]", {
              type: data.type,
              timestamp: new Date().toISOString(),
              readyState: socketRef.current.readyState,
              audioLength: data.type === "audio" ? data.audio?.length : undefined,
            });
          } catch (error) {
            console.error("❌ [WEBSOCKET SEND ERROR]", error, data);
          }
        }, []);
      
        const close = useCallback(() => {
          // ✅ تنظيف timers
          if (aiDoneTimerRef.current) {
            clearTimeout(aiDoneTimerRef.current);
            aiDoneTimerRef.current = null;
          }
      
          if (socketRef.current) {
            const readyState = socketRef.current.readyState;
            // Only close if connection is open or connecting
            if (
              readyState === WebSocket.OPEN ||
              readyState === WebSocket.CONNECTING
            ) {
              console.log("🔌 [WEBSOCKET] Closing connection, state:", readyState);
              try {
                socketRef.current.close(1000, "Normal closure");
              } catch (error) {
                console.error("❌ [WEBSOCKET] Error closing connection:", error);
              }
            } else {
              console.log(
                "ℹ️ [WEBSOCKET] Connection already closed or closing, state:",
                readyState
              );
            }
            socketRef.current = null;
          }
        }, []);
      
        const isAiSpeaking = useCallback(() => aiSpeakingRef.current, []);
      
        const getSocket = useCallback(() => socketRef.current, []);
      
        // Return stable object reference using useMemo
        return useMemo(
          () => ({
            connect,
            send,
            close,
            isAiSpeaking,
            getSocket,
          }),
          [connect, send, close, isAiSpeaking, getSocket]
        );
      }

      import React, { useState, useMemo, useEffect } from "react";
      import { useQuery } from "@tanstack/react-query";
      import {
        Card,
        Collapse,
        Typography,
        Space,
        Spin,
        Empty,
        Alert,
        Tag,
        Select,
        Row,
        Col,
        Button,
        Statistic,
      } from "antd";
      import {
        QuestionCircleOutlined,
        BulbOutlined,
        CheckCircleOutlined,
        InfoCircleOutlined,
        FilterOutlined,
        ClearOutlined,
        ReloadOutlined,
      } from "@ant-design/icons";
      import { App } from "antd";
      import useAuthStore from "../store/authStore";
      import interviewService from "../services/interviewApi";
      
      const { Title, Text, Paragraph } = Typography;
      const { Panel } = Collapse;
      const { Option } = Select;
      
      export default function InterviewPreparation() {
        const { message } = App.useApp();
        const { token, interviewQuestions, setInterviewQuestions } = useAuthStore();
        const [selectedDifficulty, setSelectedDifficulty] = useState(null);
        const [selectedCategory, setSelectedCategory] = useState(null);
      
        const {
          data: questionsData,
          isLoading,
          isError,
          error,
          refetch,
          isFetching,
        } = useQuery({
          queryKey: ["interviewQuestions"],
          queryFn: async () => {
            console.log("Fetching questions from API...");
            const result = await interviewService.getQuestions();
            console.log("Questions fetched:", result);
            return result;
          },
          enabled: false, // Don't fetch automatically
          retry: 1,
          onError: (error) => {
            console.error("Query error:", error);
          },
          onSuccess: (data) => {
            console.log("Query success:", data);
          },
        });
      
        // Use questions from store if available, otherwise from API response
        const allQuestions = useMemo(() => {
          if (interviewQuestions) {
            return (
              interviewQuestions?.data?.questions ||
              interviewQuestions?.questions ||
              interviewQuestions ||
              []
            );
          }
          return questionsData?.data?.questions || questionsData?.questions || [];
        }, [interviewQuestions, questionsData]);
      
        // Save questions to store when fetched
        useEffect(() => {
          if (questionsData && !isLoading) {
            setInterviewQuestions(questionsData);
            message.success("New interview questions generated successfully!");
          }
        }, [questionsData, isLoading, setInterviewQuestions, message]);
      
        // Handle errors
        useEffect(() => {
          if (isError && error) {
            console.error("Error fetching questions:", error);
            const errorMessage = error?.response?.data?.message || error?.message || "Failed to load interview questions";
            message.error(errorMessage);
          }
        }, [isError, error, message]);
      
        const handleGenerateNew = async () => {
          if (!token) {
            message.error("Please login to generate questions");
            return;
          }
          
          try {
            console.log("Generating new questions...");
            // Clear existing questions from store to show loading state
            setInterviewQuestions(null);
            // Trigger refetch - this will set isLoading/isFetching to true
            await refetch();
          } catch (err) {
            console.error("Error in handleGenerateNew:", err);
            message.error("Failed to generate questions. Please try again.");
          }
        };
      
        // Extract unique categories and difficulties
        const categories = useMemo(() => {
          const cats = new Set();
          allQuestions.forEach((q) => {
            if (q.category) cats.add(q.category);
          });
          return Array.from(cats).sort();
        }, [allQuestions]);
      
        const difficulties = useMemo(() => {
          const diffs = new Set();
          allQuestions.forEach((q) => {
            if (q.difficulty) diffs.add(q.difficulty);
          });
          return Array.from(diffs).sort();
        }, [allQuestions]);
      
        // Filter questions based on selected filters
        const filteredQuestions = useMemo(() => {
          return allQuestions.filter((q) => {
            const matchesDifficulty =
              !selectedDifficulty || q.difficulty === selectedDifficulty;
            const matchesCategory =
              !selectedCategory || q.category === selectedCategory;
            return matchesDifficulty && matchesCategory;
          });
        }, [allQuestions, selectedDifficulty, selectedCategory]);
      
        const getDifficultyColor = (difficulty) => {
          switch (difficulty?.toLowerCase()) {
            case "easy":
              return "green";
            case "medium":
              return "orange";
            case "hard":
              return "red";
            default:
              return "blue";
          }
        };
      
        const getCategoryColor = (category) => {
          const colors = [
            "blue",
            "purple",
            "cyan",
            "geekblue",
            "magenta",
            "volcano",
            "gold",
          ];
          const index = categories.indexOf(category);
          return colors[index % colors.length] || "default";
        };
      
        const handleClearFilters = () => {
          setSelectedDifficulty(null);
          setSelectedCategory(null);
        };
      
        const hasActiveFilters = selectedDifficulty || selectedCategory;
      
        return (
          <div>
            <Space direction="vertical" size="large" style={{ width: "100%" }}>
              <Card>
                <div
                  style={{
                    display: "flex",
                    justifyContent: "space-between",
                    alignItems: "flex-start",
                    flexWrap: "wrap",
                    gap: "16px",
                  }}
                >
                  <Space direction="vertical" size="small" style={{ flex: 1 }}>
                    <Title level={2} style={{ margin: 0, color: "#fafafa" }}>
                      Interview Preparation
                    </Title>
                    <Text style={{ color: "#a1a1aa" }}>
                      Practice with AI-generated interview questions and expert tips
                    </Text>
                    {interviewQuestions && (
                      <Text
                        style={{
                          fontSize: "12px",
                          fontStyle: "italic",
                          color: "#71717a",
                        }}
                      >
                        Showing saved questions. Click "Generate New Questions" to
                        refresh.
                      </Text>
                    )}
                  </Space>
                  <Button
                    type="primary"
                    size="large"
                    icon={<ReloadOutlined />}
                    onClick={handleGenerateNew}
                    loading={isLoading || isFetching}
                    disabled={!token || isLoading || isFetching}
                  >
                    Generate New Questions
                  </Button>
                </div>
              </Card>
      
              {(isLoading || isFetching) && (
                <Card>
                  <div style={{ textAlign: "center", padding: "40px" }}>
                    <Spin size="large" />
                    <div style={{ marginTop: "16px" }}>
                      <Text style={{ color: "#a1a1aa" }}>
                        Loading interview questions...
                      </Text>
                    </div>
                  </div>
                </Card>
              )}
      
              {isError && (
                <Alert
                  message="Error Loading Questions"
                  description={
                    error?.message ||
                    "Failed to load interview questions. Please try again later."
                  }
                  type="error"
                  showIcon
                  icon={<InfoCircleOutlined />}
                />
              )}
      
              {!isLoading &&
                !isFetching &&
                !isError &&
                allQuestions.length === 0 &&
                !interviewQuestions && (
                  <Card>
                    <Empty
                      description={
                        <Space direction="vertical" size="small">
                          <Text style={{ color: "#a1a1aa" }}>
                            No interview questions available at the moment
                          </Text>
                          <Button
                            type="primary"
                            icon={<ReloadOutlined />}
                            onClick={handleGenerateNew}
                            loading={isLoading}
                            disabled={!token}
                          >
                            Generate Questions
                          </Button>
                        </Space>
                      }
                      image={Empty.PRESENTED_IMAGE_SIMPLE}
                    />
                  </Card>
                )}
      
              {!isLoading && !isFetching && !isError && allQuestions.length > 0 && (
                <>
                  {/* Filters Section */}
                  <Card>
                    <Space
                      direction="vertical"
                      size="middle"
                      style={{ width: "100%" }}
                    >
                      <div
                        style={{
                          display: "flex",
                          justifyContent: "space-between",
                          alignItems: "center",
                          flexWrap: "wrap",
                          gap: "16px",
                        }}
                      >
                        <Space size="middle" wrap>
                          <FilterOutlined
                            style={{ fontSize: "18px", color: "#0ea5e9" }}
                          />
                          <Text strong style={{ fontSize: "16px", color: "#fafafa" }}>
                            Filter Questions
                          </Text>

                        {hasActiveFilters && (
                          <Button
                            icon={<ClearOutlined />}
                            onClick={handleClearFilters}
                            size="small"
                          >
                            Clear Filters
                          </Button>
                        )}
                      </div>
      
                      <Row gutter={[16, 16]}>
                        <Col xs={24} sm={12} lg={8}>
                          <Space
                            direction="vertical"
                            size="small"
                            style={{ width: "100%" }}

                              Difficulty

                              placeholder="All Difficulties"

                              onChange={setSelectedDifficulty}
                              allowClear
                            >
                              {difficulties.map((diff) => (
                                <Option key={diff} value={diff}>
                                  <Tag

      /**
       * InterviewReport Page
       * Full-page report viewer with navigation, print/export functionality
       * @module pages/interviewReport

      import React, { useState, useEffect } from "react";
      import { useParams, useNavigate } from "react-router-dom";
      import { Card, Button, Space, Typography, Spin, Alert, message } from "antd";
      import { ArrowLeftOutlined, PrinterOutlined, DownloadOutlined } from "@ant-design/icons";
      import InterviewReport from "../components/InterviewReport";
      import interviewService from "../services/interviewApi";
      import MainLayout from "../components/layouts/MainLayout";
      
      const { Title } = Typography;
      
      /**
       * InterviewReport Page Component
       * Displays comprehensive interview report for a specific session
       */
      export default function InterviewReportPage() {
        const { sessionId } = useParams();
        const navigate = useNavigate();
        const [report, setReport] = useState(null);
        const [loading, setLoading] = useState(true);
        const [error, setError] = useState(null);
      
        useEffect(() => {
          const fetchReport = async () => {
            if (!sessionId) {
              setError("Session ID is required");
              setLoading(false);
              return;
            }
      
            try {
              setLoading(true);
              const reportData = await interviewService.getInterviewReport(sessionId);
              setReport(reportData);
              setError(null);
            } catch (err) {
              console.error("Failed to fetch report:", err);
              setError(err.response?.data?.message || "Failed to load interview report");
              message.error("Failed to load interview report");
            } finally {
              setLoading(false);
            }
          };
      
          fetchReport();
        }, [sessionId]);
      
        const handlePrint = () => {
          window.print();
        };
      
        const handleExport = () => {
          // TODO: Implement PDF export
          message.info("Export functionality coming soon");
        };
      
        if (loading) {
          return (
            <MainLayout>
              <div style={{ padding: "24px", textAlign: "center" }}>
                <Spin size="large" />
                <div style={{ marginTop: "16px" }}>
                  <Typography.Text>Loading interview report...</Typography.Text>
                </div>
              </div>
            </MainLayout>
          );
        }
      
        if (error) {
          return (
            <MainLayout>
              <div style={{ padding: "24px" }}>
                <Alert
                  message="Error Loading Report"
                  description={error}
                  type="error"
                  showIcon
                  action={
                    <Button size="small" onClick={() => navigate("/interview-prep")}>
                      Back to Interviews
                    </Button>
                  }
                />
              </div>
            </MainLayout>
          );
        }
      
        return (
          <MainLayout>
            <div style={{ padding: "24px" }}>
              <Card>
                <Space direction="vertical" size="large" style={{ width: "100%" }}>
                  {/* Header */}
                  <Space style={{ width: "100%", justifyContent: "space-between" }}>
                    <Space>
                      <Button
                        icon={<ArrowLeftOutlined />}
                        onClick={() => navigate("/interview-prep")}
                      >
                        Back to Interviews
                      </Button>
                      <Title level={2} style={{ margin: 0 }}>
                        Interview Report
                      </Title>
                    </Space>
                    <Space>
                      <Button icon={<PrinterOutlined />} onClick={handlePrint}>
                        Print
                      </Button>
                      <Button icon={<DownloadOutlined />} onClick={handleExport}>
                        Export PDF
                      </Button>
                    </Space>
                  </Space>
      
                  {/* Report Content */}
                  <InterviewReport report={report} sessionId={sessionId} />
                </Space>
              </Card>
            </div>
          </MainLayout>
        );
      }

      import { useEffect, useState, useRef, useCallback, useMemo } from "react";
      import useAuthStore from "../store/authStore";
      import AudioVisualization from "../components/AudioVisualization";
      import UserAvatar from "../components/UserAvatar";
      import Navbar from "../components/Navbar";
      import { message } from "antd";
      import {
        EyeOutlined,
        EyeInvisibleOutlined,
        PauseOutlined,
      } from "@ant-design/icons";
      import { useMicrophone } from "../hooks/useMicrophone";
      import { useRecorder } from "../hooks/useRecorder";
      import { useWebSocket } from "../hooks/useWebSocket";
      import { playBase64Audio } from "../utils/audio";
      import "./interviewSession.css";
      
      export default function InterviewSession() {
        const { user } = useAuthStore();
        const [question] = useState("");
        const [questionCategory] = useState(null);
        const [sessionId, setSessionId] = useState(null);
        const [isStarting, setIsStarting] = useState(false);
        const [error, setError] = useState(null);
        const [showQuestion, setShowQuestion] = useState(false);
        const [evaluation] = useState(null);
        const [useVoiceMode, setUseVoiceMode] = useState(false);
        const [aiSpeaking, setAiSpeaking] = useState(false);
        const [isRecording, setIsRecording] = useState(false); // State instead of ref for UI updates
        const streamRef = useRef(null);
        const recorderRef = useRef(null);
        const runningRef = useRef(false);
        const startRecordingRef = useRef(null); // Ref for auto-start function
        const wsRef = useRef(null); // Ref for WebSocket hook
      
        // Voice Interview Hooks
        const { stream, initMic, stopMic } = useMicrophone();
      
        // Stop voice recording - defined early to be used in useWebSocket
        const handleStopVoiceRecording = useCallback(() => {
          console.log("🛑 Stopping voice recording...");
          if (recorderRef.current) {
            recorderRef.current.stopRecording();
            console.log("✅ Recorder stopped");
          }
          stopMic();
          setUseVoiceMode(false);
          setIsRecording(false);
          runningRef.current = false;
          streamRef.current = null;
          console.log("✅ Voice mode deactivated");
          message.info("Voice recording stopped");
        }, [stopMic]);
      
        // Start voice recording - defined early to be used in useWebSocket
        const handleStartVoiceRecording = useCallback(async () => {
          try {
            // Check WebSocket connection using ref
            const ws = wsRef.current;
            if (!ws) {
              console.warn(
                "⚠️ [AUTO START] WebSocket not initialized, skipping auto start"
              );
              return;
            }
      
            const wsSocket = ws.getSocket();
            if (!wsSocket || wsSocket.readyState !== WebSocket.OPEN) {
              console.warn(
                "⚠️ [AUTO START] WebSocket not connected, skipping auto start"
              );
              return;
            }
      
            // إذا كان التسجيل شغال بالفعل، لا تفعل شيء
            if (useVoiceMode && isRecording) {
              console.log("ℹ️ [AUTO START] Already recording, skipping");
              return;
            }
      
            const stream = await initMic();
            streamRef.current = stream;
      
            if (!ws.isAiSpeaking() && recorderRef.current) {
              recorderRef.current.startRecording(stream);
            }
      
            setUseVoiceMode(true);
            setIsRecording(true);
            runningRef.current = true;
            console.log("✅ [AUTO START] Voice recording started automatically");
          } catch (error) {
            console.error("❌ [AUTO START] Error starting voice recording:", error);
            // لا نعرض error message في حالة auto start
          }
        }, [initMic, useVoiceMode, isRecording]);
      
        // WebSocket hook - handles all communication
        const ws = useWebSocket(
          (audio) => {
            // استقبال صوت AI
            console.log("🔊 [AUDIO RECEIVED] Playing AI audio");
            setAiSpeaking(true); // تحديث state فوراً
            if (recorderRef.current) {
              recorderRef.current.pauseRecording();
            }
            // تشغيل الصوت
            playBase64Audio(audio)
              .then(() => {
                console.log("✅ [AUDIO PLAYBACK] Finished");
                // لا نحدث state هنا - سيتم تحديثه من onAiDone
                if (runningRef.current && stream && recorderRef.current) {
                  console.log(
                    "🔄 [AUTO RESTART] Restarting recording after AI finished"
                  );
                  // لا نستأنف هنا - سيتم من onAiDone
                }
              })
              .catch((err) => {
                console.error("❌ [AUDIO PLAYBACK ERROR]", err);
                setAiSpeaking(false);
                if (isRecording && stream && recorderRef.current) {
                  recorderRef.current.resumeRecording();
                }
              });
          },
          () => {
            // عند انتهاء كلام AI
            console.log("✅ [AI DONE] AI finished speaking");
            setAiSpeaking(false); // تحديث state
      
            // بدء التسجيل تلقائياً إذا لم يكن شغال
            if (!useVoiceMode && sessionId) {
              console.log(
                "🎙️ [AUTO START] Starting voice mode automatically after AI finished"
              );
              // Use ref to avoid circular dependency
              if (startRecordingRef.current) {
                startRecordingRef.current();
              }
            } else if (isRecording && stream && recorderRef.current) {
              console.log("🔄 [AUTO RESTART] Resuming recording after AI finished");
              recorderRef.current.resumeRecording();
            }
          },
          () => {
            // End ACK - recorder handles this
            if (recorderRef.current) {
              recorderRef.current.onEndAcknowledged();
            }
          },
          () => {
            // Interview ended
            console.log("🏁 [INTERVIEW ENDED] Cleaning up...");
            handleStopVoiceRecording();
          }
        );
      
        // Recorder hook - sends audio chunks and commit signals
        const recorder = useRecorder(
          (data) => {
            // Send audio chunk or commit signal via WebSocket
            if (wsRef.current) {
              wsRef.current.send(data);
            }
          },
          () => aiSpeaking
        );
      
        // ✅ مستوى الصوت من recorder
        const voiceSensitivity = recorder.volumeLevel || 0;
      
        // Save references
        useEffect(() => {
          recorderRef.current = recorder;
          wsRef.current = ws;
        }, [recorder, ws]);
      
        useEffect(() => {
          // Session is active when WebSocket is connected
          const wsSocket = ws.getSocket();
          const isConnected = wsSocket && wsSocket.readyState === WebSocket.OPEN;
          runningRef.current = useVoiceMode && isConnected;
        }, [useVoiceMode, ws]);
      
        // Transcription from user (voice mode)
        const [transcript] = useState("");
      
        // Handle starting interview - WebSocket only
        const handleStartInterview = async () => {
          if (!user?._id) {
            setError("Please login first.");
            message.error("Please login first to start an interview");
            return;
          }
      
          setIsStarting(true);
          setError(null);
          setSessionId(null);
          localStorage.removeItem("activeInterviewSessionId");
      
          console.log("🚀 [INTERVIEW] Starting interview...");
      
          try {
            // Connect WebSocket - connection = session started
            console.log("🔌 [WEBSOCKET] Connecting to start interview...");
            ws.connect();
      
            // Wait for WebSocket connection
            await new Promise((resolve, reject) => {
              const timeout = setTimeout(() => {
                reject(
                  new Error("Connection timeout - server may be down or unreachable")
                );
              }, 10000);
      
              // Check connection status via useWebSocket
              const checkConnection = setInterval(() => {
                const wsSocket = ws.getSocket();
                if (wsSocket && wsSocket.readyState === WebSocket.OPEN) {
                  clearInterval(checkConnection);
                  clearTimeout(timeout);
                  console.log("✅ [WEBSOCKET] Connected - Session started");
                  setSessionId("active"); // Session is active when WebSocket is connected
                  setIsStarting(false);
                  resolve();
                } else if (wsSocket && wsSocket.readyState === WebSocket.CLOSED) {
                  clearInterval(checkConnection);
                  clearTimeout(timeout);
                  reject(new Error("WebSocket connection closed"));
                }
              }, 100);
      
              // Cleanup on timeout
              setTimeout(() => {
                clearInterval(checkConnection);
              }, 10000);
            });
          } catch (err) {
            const errorMsg = err.message || "Failed to start interview session";
            setError(errorMsg);
            message.error(errorMsg);
            setSessionId(null);
            localStorage.removeItem("activeInterviewSessionId");
            setIsStarting(false);
            ws.close();
          }
        };
      
        // Cleanup on unmount - empty deps to avoid closing on every render
        useEffect(() => {
          return () => {
            console.log("🧹 [CLEANUP] Cleaning up interview session...");
      
            // Stop voice recording first
            handleStopVoiceRecording();
      
            // Close WebSocket connection
            ws.close();
          };
          // eslint-disable-next-line react-hooks/exhaustive-deps
        }, []); // Empty deps - only run on unmount
      
        const userName = user?.name || user?.email?.split("@")[0] || "User";
      
        // Use both state and hook to ensure UI updates
        const isAISpeaking = aiSpeaking || ws.isAiSpeaking();
      
        // Calculate isUserSpeaking - use useMemo to avoid ref access during render
        const isUserSpeaking = useMemo(() => {
          if (!useVoiceMode) return false;
          const wsSocket = ws.getSocket();
          const isConnected = wsSocket && wsSocket.readyState === WebSocket.OPEN;
          const aiIsSpeaking = aiSpeaking || ws.isAiSpeaking();
          // User is speaking when voice mode is on, connected, AI is not speaking, and recording is active
          return isConnected && !aiIsSpeaking && isRecording;
        }, [useVoiceMode, aiSpeaking, isRecording, ws]);
      
        // Manual start voice recording (with user interaction)
        const handleStartVoiceRecordingManual = useCallback(async () => {
          try {
            // Check WebSocket connection
            const wsSocket = ws.getSocket();
            if (!wsSocket || wsSocket.readyState !== WebSocket.OPEN) {
              setError("Not connected to server");
              message.error("Not connected to server. Please start interview first.");
              return;
            }
      
            await handleStartVoiceRecording();
            message.success("Voice recording started");
          } catch (error) {
            console.error("❌ Error starting voice recording:", error);
            setError(error.message || "Failed to start voice recording");
            message.error(error.message || "Failed to start voice recording");
          }
        }, [handleStartVoiceRecording, ws]);
      
        // End interview
        const handleEndInterview = useCallback(() => {
          // Stop voice recording if active
          if (useVoiceMode && recorderRef.current) {
            recorderRef.current.stopRecording();
            stopMic();
            setUseVoiceMode(false);
            runningRef.current = false;
            streamRef.current = null;
          }
      
          // Close WebSocket connection
          ws.close();
      
          setSessionId(null);
          localStorage.removeItem("activeInterviewSessionId");
          message.info("Interview ended");
        }, [useVoiceMode, stopMic, ws]);
      
        return (
          <div className="interview-container">
            <Navbar />
      
            {/* Error Banner */}
            {error && (
              <div className="interview-error-banner">
                <strong>Error:</strong> {error}
              </div>
            )}
      
            {/* Start Screen */}
            {!sessionId && !isStarting && (
              <div className="interview-start-screen">
                <div className="start-screen-card">
                  <span className="start-screen-icon">🎙️</span>
                  <h1 className="start-screen-title">AI Interview</h1>
                  <p className="start-screen-subtitle">
                    Practice your interview skills with our AI interviewer. Get
                    real-time feedback and improve your responses.
                  </p>
                  <button
                    className="start-interview-btn"
                    onClick={handleStartInterview}
                    disabled={isStarting || !user?._id}
                  >
                    {isStarting ? (
                      <>
                        <span className="loading-spinner" />
                        Starting...
                      </>
                    ) : (
                      <>
                        <VideoCameraIcon />
                        Start Interview
                      </>
                    )}
                  </button>
                  {!user?._id && (
                    <p
                      style={{
                        marginTop: "16px",
                        color: "#f87171",
                        fontSize: "14px",
                      }}
                    >
                      Please login first to start an interview
                    </p>
                  )}
                </div>
              </div>
            )}
      
            {/* Interview Room */}
            {(sessionId || isStarting) && (
              <div className="interview-meeting-room">
                {/* Header */}
                <div className="meeting-header">
                  <div className="meeting-title">
                    <h2>Live Interview</h2>
                    <div className="connection-badge">
                      <span

                            ws.getSocket()?.readyState === WebSocket.OPEN

      // Helper function to ensure URLs are absolute
      const normalizeUrl = (url) => {
        if (!url) return url;
        if (url.startsWith("http://") || url.startsWith("https://")) {
          return url;
        }
        return `https://${url}`;
      };
      
      export default function Profile() {
        const { message } = App.useApp();
        const { documentData, token, updateDocumentData } = useAuthStore();
        const [fileList, setFileList] = useState([]);
        const [jobId, setJobId] = useState(null);
        const [jobStatus, setJobStatus] = useState(null);
        const [jobProgress, setJobProgress] = useState(0);
        const pollingIntervalRef = useRef(null);
      
        // Check if document is uploaded by verifying fileName or fileType exists
        const hasDocument =
          documentData && (documentData.fileName || documentData.fileType);
        const [showUpload, setShowUpload] = useState(!hasDocument);
      
        // Fetch document data on component mount
        const { isLoading: isLoadingDocument } = useQuery({
          queryKey: ["document", token],
          queryFn: () => getDocument(token),
          enabled: !!token && !documentData,
          retry: 1,
          onSuccess: (data) => {
            if (data.status === "success" && data.data && data.data.document) {
              updateDocumentData(data.data.document);
            }
          },
          onError: (error) => {
            // Silently fail if no document exists
            console.log("No document found or error:", error.message);
          },
        });
      
        // Polling function to check job status
        const pollJobStatus = useCallback(
          async (currentJobId) => {
            if (!currentJobId || !token) return;
      
            try {
              const statusData = await getUploadStatus(currentJobId, token);
      
              if (statusData.status === "success" && statusData.data) {
                const { status, progress, document } = statusData.data;
      
                setJobStatus(status);
                setJobProgress(progress || 0);
      
                if (status === "completed") {
                  // Stop polling
                  if (pollingIntervalRef.current) {
                    clearInterval(pollingIntervalRef.current);
                    pollingIntervalRef.current = null;
                  }
      
                  // Fetch the updated document data from the server
                  try {
                    const documentResponse = await getDocument(token);
                    if (
                      documentResponse.status === "success" &&
                      documentResponse.data &&
                      documentResponse.data.document
                    ) {
                      updateDocumentData(documentResponse.data.document);
                      console.log(
                        "Document data updated:",
                        documentResponse.data.document
                      );
                    } else if (document) {
                      // Fallback: use document from status response if available
                      updateDocumentData(document);
                    }
                  } catch (docError) {
                    console.error(
                      "Error fetching document after completion:",
                      docError
                    );
                    // Fallback: use document from status response if available
                    if (document) {
                      updateDocumentData(document);
                    }
                  }
      
                  setFileList([]);
                  setShowUpload(false);
                  setJobId(null);
                  setJobStatus(null);
                  setJobProgress(0);
                  message.success("Document processed successfully!");
                } else if (status === "failed") {
                  // Stop polling
                  if (pollingIntervalRef.current) {

      // لازم AudioContext يكون بنفس الـ sample rate عشان الصوت يطلع بالسرعة الصحيحة

      let nextPlayTime = 0;
      
      // ✅ الحصول على الوقت المتبقي حتى ينتهي الصوت
      export function getRemainingAudioTime() {
        if (nextPlayTime === 0) return 0;
        const now = audioContext.currentTime;
        const remaining = nextPlayTime - now;
        return remaining > 0 ? remaining * 1000 : 0; // تحويل إلى milliseconds
      }
      
      export function bufferToBase64(buffer) {
        const bytes = new Uint8Array(buffer);
        let binary = "";
        for (let i = 0; i < bytes.byteLength; i++) {
          binary += String.fromCharCode(bytes[i]);
        }
        return btoa(binary);
      }
      
      export async function playBase64Audio(base64) {
        // 🔥 ده المهم في Vite / Chrome
        if (audioContext.state === "suspended") {
          await audioContext.resume();
        }
      
        // البيانات دائماً PCM خام من GPT-Realtime
        // لا نحتاج decodeAudioData
        playPCMAudio(base64);
      }
      
      function playPCMAudio(base64) {
        const binary = atob(base64);
        const bytes = new Uint8Array(binary.length);
        for (let i = 0; i < binary.length; i++) {
          bytes[i] = binary.charCodeAt(i);
        }
      
        // PCM 16-bit little-endian: كل sample = 2 bytes
        const numSamples = bytes.length / 2;
      
        // معدل العينة: 24000 Hz (نفس GPT-Realtime)
        const sampleRate = 24000;
      
        const audioBuffer = audioContext.createBuffer(1, numSamples, sampleRate);
        const channelData = audioBuffer.getChannelData(0);
      
        // قراءة PCM 16-bit little-endian
        for (let i = 0; i < numSamples; i++) {
          const byteIndex = i * 2;
          // Little-endian: البايت الأول هو الأقل أهمية
          const low = bytes[byteIndex];
          const high = bytes[byteIndex + 1];
      
          // دمج البايتين في sample واحد (16-bit signed integer)
          let sample = (high << 8) | low;
      
          // تحويل إلى signed integer (two's complement)
          if (sample >= 0x8000) {
            sample -= 0x10000;
          }
      
          // تحويل إلى float بين -1.0 و 1.0
          channelData[i] = sample / 32768.0;
        }
      
        const source = audioContext.createBufferSource();
        source.buffer = audioBuffer;
        source.connect(audioContext.destination);
        source.playbackRate.value = 1.0;
      
        // 🔥 Audio Scheduling: تحديد متى يجب تشغيل هذا الـ chunk
        const now = audioContext.currentTime;
        if (nextPlayTime < now) {
          nextPlayTime = now;
        }
      
        source.start(nextPlayTime);
      
        // تحديث nextPlayTime ليكون بعد انتهاء هذا الـ chunk
        const duration = audioBuffer.duration;
        nextPlayTime += duration;
      }

      export const testRestApiConnection = async (token) => {
        const result = {
          success: false,
          error: null,
          status: null,
          message: "",

        try {
          const response = await fetch(`${BASE_URL}/api/v1/interview/questions/sessions`, {
            method: "GET",
            headers: {
              Authorization: `Bearer ${token}`,
              "Content-Type": "application/json",
            },

          result.status = response.status;
          result.success = response.ok;
      
          if (response.ok) {
            result.message = "REST API connection successful";
          } else {
            const errorData = await response.json().catch(() => ({}));
            result.error = errorData.message || `HTTP ${response.status}`;
            result.message = `REST API connection failed: ${result.error}`;
          }
        } catch (error) {
          result.error = error.message;
          result.message = `REST API connection error: ${error.message}`;
        }
      
        return result;
      };
      
      /**
       * Test Socket.io connection
       * @param {string} token - JWT token
       * @returns {Promise<object>} - Test result
       */
      export const testSocketConnection = (token) => {
        return new Promise((resolve) => {
          const result = {
            success: false,
            error: null,
            message: "",
            socketId: null,
          };
      
          const socket = io(BASE_URL, {
            auth: {
              token: token,
            },
            transports: ["websocket", "polling"],
            timeout: 5000,
          });
      
          const timeout = setTimeout(() => {
            socket.disconnect();
            result.error = "Connection timeout";
            result.message = "Socket connection timed out after 5 seconds";
            resolve(result);
          }, 5000);
      
          socket.on("connect", () => {
            clearTimeout(timeout);
            result.success = true;
            result.socketId = socket.id;
            result.message = "Socket connection successful";
            socket.disconnect();
            resolve(result);
          });
      
          socket.on("connect_error", (error) => {
            clearTimeout(timeout);
            result.error = error.message;
            result.message = `Socket connection failed: ${error.message}`;
            socket.disconnect();
            resolve(result);
          });
      
          socket.on("disconnect", () => {
            // Connection test complete
          });
        });
      };

      export const runConnectionDiagnostics = async (token) => {
        const report = {
          timestamp: new Date().toISOString(),
          tokenInfo: getTokenInfo(token),
          restApiTest: null,
          socketTest: null,
          summary: {
            allTestsPassed: false,
            issues: [],
            recommendations: [],

        // Ensure summary is always initialized
        if (!report.summary) {
          report.summary = {
            allTestsPassed: false,
            issues: [],
            recommendations: [],
          };
        }
      
        // Token validation
        if (!token) {
          report.summary.issues.push("Token is missing");
          report.summary.recommendations.push("Please log in to get a valid token");
          return report;
        }
      
        if (!validateToken(token)) {
          report.summary.issues.push("Token has invalid format");
          report.summary.recommendations.push("Token should be a valid JWT with 3 parts");
          return report;
        }
      
        if (isTokenExpired(token)) {
          report.summary.issues.push("Token is expired");
          report.summary.recommendations.push("Please log in again to get a fresh token");
        }
      
        // Test REST API
        console.log("Testing REST API connection...");
        report.restApiTest = await testRestApiConnection(token);
        if (!report.restApiTest.success) {
          report.summary.issues.push(`REST API: ${report.restApiTest.error}`);

            report.summary.recommendations.push("Token may be invalid or expired");
          } else if (report.restApiTest.status === 0) {
            report.summary.recommendations.push("Server may not be running");
          }
        }
      
        // Test Socket.io
        console.log("Testing Socket.io connection...");
        report.socketTest = await testSocketConnection(token);
        if (!report.socketTest.success) {
          report.summary.issues.push(`Socket.io: ${report.socketTest.error}`);

            report.summary.recommendations.push("Server may not be running or not responding");
          } else if (report.socketTest.error?.includes("authentication")) {
            report.summary.recommendations.push("Token authentication failed on server");
          }
        }
      
        // Summary
        report.summary.allTestsPassed =
          report.restApiTest.success && report.socketTest.success;
      
        if (report.summary.allTestsPassed) {
          report.summary.recommendations.push("All connection tests passed!");
        }
      
        return report;
      };

      /**
       * Token validation and debugging utilities
       */
      
      /**
       * Check if token exists and has valid format
       * @param {string} token - JWT token
       * @returns {boolean} - True if token exists and has valid format
       */
      export const validateToken = (token) => {
        if (!token || typeof token !== "string") {
          return false;
        }
      
        // JWT tokens have 3 parts separated by dots
        const parts = token.split(".");
        if (parts.length !== 3) {
          return false;
        }
      
        // Check if parts are base64 encoded
        try {
          parts.forEach((part) => {
            if (part.length === 0) {
              throw new Error("Empty token part");
            }
            // Basic base64 check
            atob(part);
          });
          return true;
        } catch (e) {
          return false;
        }
      };
      
      /**
       * Decode JWT payload without verification
       * @param {string} token - JWT token
       * @returns {object|null} - Decoded payload or null if invalid
       */
      export const decodeToken = (token) => {
        if (!validateToken(token)) {
          return null;
        }
      
        try {
          const parts = token.split(".");
          const payload = parts[1];
          const decoded = JSON.parse(atob(payload));
          return decoded;
        } catch (e) {
          console.error("Error decoding token: "***"No token";
        return `${token.substring(0, 20)}...`;
      };
